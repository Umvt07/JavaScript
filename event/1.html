/* ======================== JAVASCRIPT EVENT OBJECT ======================== Whenever an event happens (click, keypress, submit, etc.) the browser creates an EVENT OBJECT. Example: document.addEventListener('click', function(e){ console.log(e); }); Here: e = event object -------------------------------------------------- 1. type -------------------------------------------------- Tells which event happened. Example: e.type Output: "click" "keydown" "submit" -------------------------------------------------- 2. timeStamp -------------------------------------------------- Time when the event occurred. Example: e.timeStamp Output: 2456.7 (milliseconds since page load) -------------------------------------------------- 3. defaultPrevented -------------------------------------------------- Checks whether preventDefault() was used. Example: form.addEventListener('submit', function(e){ e.preventDefault(); console.log(e.defaultPrevented); }); Output: true -------------------------------------------------- 4. target -------------------------------------------------- The ACTUAL element on which event happened. Example: document.getElementById('btn').addEventListener('click', function(e){ console.log(e.target); }); Output: Very important property. -------------------------------------------------- 5. currentTarget -------------------------------------------------- The element whose listener is currently running. Example:
document.getElementById('parent').addEventListener('click', function(e){ console.log(e.target.id); console.log(e.currentTarget.id); }); If button clicked: target = child currentTarget = parent Difference: target -> where event actually happened currentTarget -> where listener is attached -------------------------------------------------- 6. srcElement -------------------------------------------------- Old Internet Explorer version of target. Use: e.target instead of: e.srcElement -------------------------------------------------- 7. toElement -------------------------------------------------- Used in old mouse events. Represents element mouse moved into. Rarely used today. Modern alternative: relatedTarget -------------------------------------------------- 8. clientX and clientY -------------------------------------------------- Mouse position relative to browser window. Example: document.addEventListener('click', function(e){ console.log(e.clientX); console.log(e.clientY); }); Output: 320 180 Meaning: 320px from left 180px from top -------------------------------------------------- 9. screenX and screenY -------------------------------------------------- Mouse position relative to entire computer screen. Example: e.screenX e.screenY Difference: clientX/Y -> browser window coordinates screenX/Y -> monitor screen coordinates -------------------------------------------------- 10. altKey -------------------------------------------------- Checks whether ALT key was pressed. Example: document.addEventListener('click', function(e){ console.log(e.altKey); }); Output: true / false -------------------------------------------------- 11. ctrlKey -------------------------------------------------- Checks whether CTRL key pressed. Example: e.ctrlKey Useful for shortcuts. -------------------------------------------------- 12. shiftKey -------------------------------------------------- Checks whether SHIFT key pressed. Example: e.shiftKey -------------------------------------------------- 13. keyCode -------------------------------------------------- Numeric code of pressed key. Example: document.addEventListener('keydown', function(e){ console.log(e.keyCode); }); Enter -> 13 A -> 65 Deprecated now. Modern way: e.key Example: document.addEventListener('keydown', function(e){ console.log(e.key); }); Outputs: "Enter" "a" "ArrowUp" -------------------------------------------------- MOST IMPORTANT PROPERTIES -------------------------------------------------- e.target -> which element clicked e.currentTarget -> where listener attached e.preventDefault() -> stop default browser action e.clientX / e.clientY -> mouse position e.key -> keyboard key e.ctrlKey / e.shiftKey -> shortcut detection -------------------------------------------------- REAL EXAMPLE -------------------------------------------------- document.getElementById('btn').addEventListener('click', function(e){ console.log("Type:", e.type); console.log("Target:", e.target); console.log("X:", e.clientX); console.log("Y:", e.clientY); console.log("Ctrl pressed:", e.ctrlKey); }); -------------------------------------------------- SHORTCUT EXAMPLE -------------------------------------------------- document.addEventListener('keydown', function(e){ if(e.ctrlKey && e.key === 's'){ e.preventDefault(); console.log("Save shortcut pressed"); } }); Detects: CTRL + S -------------------------------------------------- IMPORTANT NOTE -------------------------------------------------- Different events have different properties. Mouse events: clientX, clientY Keyboard events: key, keyCode Submit events: form related properties So every property does NOT exist in every event. */
